"use client";

import { useEffect, useState } from "react";

export default function Dashboard() {
  const [emails, setEmails] = useState<any[]>([]);
  const [loading, setLoading] = useState(false);
  const [filter, setFilter] = useState("all");
  const [search, setSearch] = useState("");

  const API = "https://ai-email-copilot-production.up.railway.app";

  // ✅ Load data on page load
  useEffect(() => {
    loadHistory();
  }, []);

  // 📥 Load from DB
  const loadHistory = async () => {
    try {
      setLoading(true);
      const res = await fetch(`${API}/api/history`);
      const data = await res.json();
      console.log("History:", data);
      setEmails(data || []);
    } catch (err) {
      console.error("History error:", err);
    } finally {
      setLoading(false);
    }
  };

  // 🔄 Sync new emails (IMAP → DB)
  const syncEmails = async () => {
    try {
      setLoading(true);
      await fetch(`${API}/api/fetch-and-analyze`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({}),
      });

      // reload updated data
      await loadHistory();
    } catch (err) {
      console.error("Sync error:", err);
    } finally {
      setLoading(false);
    }
  };

  // 🔍 Filter logic
  const filteredEmails = emails.filter((e: any) => {
    if (filter !== "all" && e.priority !== filter) return false;
    if (!e.summary?.toLowerCase().includes(search.toLowerCase())) return false;
    return true;
  });

  const urgent = emails.filter((e) => e.priority === "urgent").length;
  const normal = emails.filter((e) => e.priority === "normal").length;
  const totalTasks = emails.flatMap((e: any) => e.tasks || []).length;
  const emailsWithReplies = emails.filter((e: any) => e.suggested_reply).length;

  return (
    <div className="flex h-screen bg-gray-50">
      {/* Sidebar */}
      <div className="w-64 bg-white shadow-md p-6 border-r border-gray-200">
        <h2 className="text-2xl font-bold mb-8">📧 Copilot</h2>

        <button
          onClick={syncEmails}
          disabled={loading}
          className="w-full bg-blue-600 hover:bg-blue-700 text-white py-3 px-4 rounded-lg mb-6 font-medium transition disabled:opacity-50 disabled:cursor-not-allowed"
        >
          {loading ? "⏳ Processing..." : "🔄 Sync Emails"}
        </button>

        <div className="space-y-2">
          <FilterButton
            label="📊 All"
            value="all"
            setFilter={setFilter}
            active={filter}
          />
          <FilterButton
            label="🔴 Urgent"
            value="urgent"
            setFilter={setFilter}
            active={filter}
          />
          <FilterButton
            label="⚪ Normal"
            value="normal"
            setFilter={setFilter}
            active={filter}
          />
        </div>
      </div>

      {/* Main Content */}
      <div className="flex-1 p-8 overflow-auto">
        {/* Header */}
        <div className="flex justify-between items-center mb-8">
          <h1 className="text-3xl font-bold text-gray-900">Dashboard</h1>

          <input
            type="text"
            placeholder="Search emails..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="border border-gray-300 px-4 py-2 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500 w-64"
          />
        </div>

        {/* Stats */}
        <div className="grid grid-cols-5 gap-6 mb-8">
          <StatCard title="Total Emails" value={emails.length} />
          <StatCard title="🔴 Urgent" value={urgent} />
          <StatCard title="⚪ Normal" value={normal} />
          <StatCard title="📋 Tasks" value={totalTasks} />
          <StatCard title="💬 Replies" value={emailsWithReplies} />
        </div>

        {/* Loading State */}
        {loading && (
          <div className="text-center py-8">
            <p className="text-gray-600 text-lg">⏳ Loading...</p>
          </div>
        )}

        {/* Content Grid */}
        {!loading && (
          <div className="grid grid-cols-3 gap-8">
            {/* Emails Section */}
            <div className="bg-white rounded-lg shadow-md p-6">
              <h2 className="text-xl font-semibold mb-6 flex items-center">
                📩 Emails
              </h2>

              {filteredEmails.length === 0 ? (
                <p className="text-gray-500 text-center py-8">
                  No emails found
                </p>
              ) : (
                <div className="space-y-3 max-h-96 overflow-y-auto">
                  {filteredEmails.map((email: any, i: number) => (
                    <div
                      key={i}
                      className="bg-gray-50 p-4 rounded-lg border border-gray-200 hover:shadow-md hover:border-blue-300 transition cursor-pointer"
                    >
                      {/* Email Header */}
                      <div className="mb-3">
                        <div className="flex justify-between items-start mb-2">
                          <div className="flex-1">
                            <p className="font-semibold text-gray-900 text-sm">
                              📧 {email.email_from || "Unknown Sender"}
                            </p>
                            <p className="text-gray-700 font-medium text-sm mt-1">
                              {email.email_subject || "No Subject"}
                            </p>
                          </div>
                          <span
                            className={`inline-block text-xs font-semibold px-2 py-1 rounded-full ml-2 ${
                              email.priority === "urgent"
                                ? "bg-red-100 text-red-700"
                                : "bg-gray-200 text-gray-700"
                            }`}
                          >
                            {email.priority || "normal"}
                          </span>
                        </div>
                      </div>

                      {/* Summary */}
                      <p className="text-gray-800 mb-3 line-clamp-2">
                        {email.summary?.slice(0, 150)}...
                      </p>

                      {/* Key Points */}
                      {email.key_points && email.key_points.length > 0 && (
                        <div className="mb-3">
                          <p className="text-xs font-semibold text-gray-600 mb-1">🔑 Key Points:</p>
                          <ul className="text-xs text-gray-700 space-y-1">
                            {email.key_points.slice(0, 2).map((point: string, idx: number) => (
                              <li key={idx} className="flex items-start">
                                <span className="text-blue-500 mr-1">•</span>
                                <span className="line-clamp-1">{point}</span>
                              </li>
                            ))}
                          </ul>
                        </div>
                      )}

                      {/* Suggested Reply Preview */}
                      {email.suggested_reply && (
                        <div className="mb-3">
                          <p className="text-xs font-semibold text-gray-600 mb-1">💬 Suggested Reply:</p>
                          <p className="text-xs text-gray-700 bg-blue-50 p-2 rounded line-clamp-2">
                            {email.suggested_reply.slice(0, 100)}...
                          </p>
                        </div>
                      )}

                      {/* Tasks Count */}
                      {email.tasks && email.tasks.length > 0 && (
                        <div className="flex items-center text-xs text-gray-600">
                          <span className="mr-1">📋</span>
                          <span>{email.tasks.length} task{email.tasks.length !== 1 ? 's' : ''}</span>
                        </div>
                      )}
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Tasks Section */}
            <div className="bg-white rounded-lg shadow-md p-6">
              <h2 className="text-xl font-semibold mb-6 flex items-center">
                📋 Tasks
              </h2>

              {filteredEmails.flatMap((e: any) => e.tasks || []).length ===
              0 ? (
                <p className="text-gray-500 text-center py-8">No tasks</p>
              ) : (
                <div className="space-y-3 max-h-96 overflow-y-auto">
                  {filteredEmails
                    .flatMap((e: any) => e.tasks || [])
                    .map((task: any, i: number) => (
                      <div
                        key={i}
                        className="bg-gray-50 p-4 rounded-lg border border-gray-200 hover:shadow-md transition"
                      >
                        <div className="flex items-start justify-between mb-2">
                          <p className="text-gray-900 font-medium flex-1">
                            {task.task || task.description || "Task"}
                          </p>
                          {task.priority && (
                            <span
                              className={`inline-block text-xs font-semibold px-2 py-1 rounded-full ml-2 ${
                                task.priority === "urgent" || task.priority === "high"
                                  ? "bg-red-100 text-red-700"
                                  : task.priority === "medium"
                                  ? "bg-yellow-100 text-yellow-700"
                                  : "bg-gray-200 text-gray-700"
                              }`}
                            >
                              {task.priority}
                            </span>
                          )}
                        </div>

                        <div className="flex items-center justify-between text-sm text-gray-600">
                          <div className="flex items-center space-x-4">
                            {task.deadline && (
                              <span className="flex items-center">
                                ⏰ {task.deadline}
                              </span>
                            )}
                            {task.status && (
                              <span className={`px-2 py-1 rounded text-xs font-medium ${
                                task.status === "completed"
                                  ? "bg-green-100 text-green-700"
                                  : task.status === "in_progress"
                                  ? "bg-blue-100 text-blue-700"
                                  : "bg-gray-100 text-gray-700"
                              }`}>
                                {task.status}
                              </span>
                            )}
                          </div>
                        </div>

                        {task.assigned_to && (
                          <p className="text-xs text-gray-500 mt-2">
                            👤 Assigned to: {task.assigned_to}
                          </p>
                        )}
                      </div>
                    ))}
                </div>
              )}
            </div>

            {/* Suggested Replies Section */}
            <div className="bg-white rounded-lg shadow-md p-6">
              <h2 className="text-xl font-semibold mb-6 flex items-center">
                💬 Suggested Replies
              </h2>

              {filteredEmails.filter((e: any) => e.suggested_reply).length === 0 ? (
                <p className="text-gray-500 text-center py-8">No suggested replies</p>
              ) : (
                <div className="space-y-3 max-h-96 overflow-y-auto">
                  {filteredEmails
                    .filter((e: any) => e.suggested_reply)
                    .map((email: any, i: number) => (
                      <div
                        key={i}
                        className="bg-blue-50 p-4 rounded-lg border border-blue-200"
                      >
                        <div className="mb-3">
                          <p className="text-sm font-medium text-gray-700 mb-1">
                            📧 From: {email.email_from || "Unknown"}
                          </p>
                          <p className="text-sm text-gray-600 line-clamp-1">
                            {email.email_subject || "No Subject"}
                          </p>
                        </div>

                        <div className="bg-white p-3 rounded border-l-4 border-blue-400">
                          <p className="text-sm text-gray-800 line-clamp-3">
                            {email.suggested_reply}
                          </p>
                        </div>

                        <div className="flex items-center justify-between mt-3">
                          <span
                            className={`inline-block text-xs font-semibold px-2 py-1 rounded-full ${
                              email.priority === "urgent"
                                ? "bg-red-100 text-red-700"
                                : "bg-gray-200 text-gray-700"
                            }`}
                          >
                            {email.priority || "normal"}
                          </span>
                          <button className="text-xs text-blue-600 hover:text-blue-800 font-medium">
                            Copy Reply
                          </button>
                        </div>
                      </div>
                    ))}
                </div>
              )}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}

/* Components */

function StatCard({ title, value }: { title: string; value: number }) {
  return (
    <div className="bg-white rounded-lg shadow-md p-6 border-l-4 border-blue-500">
      <p className="text-gray-600 text-sm font-medium mb-2">{title}</p>
      <p className="text-4xl font-bold text-gray-900">{value}</p>
    </div>
  );
}

function FilterButton({
  label,
  value,
  setFilter,
  active,
}: {
  label: string;
  value: string;
  setFilter: (value: string) => void;
  active: string;
}) {
  return (
    <button
      onClick={() => setFilter(value)}
      className={`w-full text-left px-4 py-2 rounded-lg font-medium transition ${
        active === value
          ? "bg-blue-100 text-blue-600"
          : "text-gray-700 hover:bg-gray-100"
      }`}
    >
      {label}
    </button>
  );
}
